# Git Commands - Copy & Paste Ready
# Просто скопіюйте та вставте ці команди в термінал

# ==========================================
# ВАРІАНТ 1: Додати всі файли одразу
# ==========================================

git add .


# ==========================================
# ВАРІАНТ 2: Додати файли окремо (рекомендовано)
# ==========================================

# Нові файли - Код
git add src/types/chat.ts
git add src/services/openrouter.ts
git add src/components/ChatMessage.tsx
git add src/pages/AIChatPage.tsx

# Змінені файли - Код
git add src/App.tsx
git add src/pages/HomePage.tsx

# Конфігурація
git add .env.example
git add .gitignore

# Документація
git add AI_CHAT_README.md
git add AI_CHAT_QUICKSTART.md
git add AI_CHAT_SETUP.md
git add AI_CHAT_FILES.txt
git add OPENROUTER_SETUP_GUIDE.md
git add GIT_BRANCH_COMMANDS.md
git add INTEGRATION_COMPLETE.md
git add git-add-commands.txt


# ==========================================
# ПОВНА ПОСЛІДОВНІСТЬ КОМАНД
# ==========================================

# 1. Створити гілку
git checkout -b feature/ai-chat-integration
